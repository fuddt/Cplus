#include <cstdint>
#include <vector>

std::vector<std::uint8_t> resizeNearest(
    const std::vector<std::uint8_t>& small,
    int smallRows,
    int smallCols,
    int bigRows,
    int bigCols
)
{
    std::vector<std::uint8_t> big(
        static_cast<std::size_t>(bigRows * bigCols)
    );

    for (int y = 0; y < bigRows; ++y)
    {
        // big側のyがsmall側のどこに対応するか
        int srcY = y * smallRows / bigRows;

        for (int x = 0; x < bigCols; ++x)
        {
            // big側のxがsmall側のどこに対応するか
            int srcX = x * smallCols / bigCols;

            big[
                static_cast<std::size_t>(y * bigCols + x)
            ] =
                small[
                    static_cast<std::size_t>(srcY * smallCols + srcX)
                ];
        }
    }

    return big;
}



#include <cstdint>
#include <vector>

std::vector<std::uint8_t> resize1DNearest(
    const std::vector<std::uint8_t>& small,
    int smallSize,
    int bigSize
)
{
    std::vector<std::uint8_t> big(static_cast<std::size_t>(bigSize));

    for (int i = 0; i < bigSize; ++i)
    {
        // big側の i 番目が、small側のどこに対応するかを求める
        // int同士の割り算なので小数は切り捨て
        int srcIndex = i * smallSize / bigSize;

        big[static_cast<std::size_t>(i)] =
            small[static_cast<std::size_t>(srcIndex)];
    }

    return big;
}
