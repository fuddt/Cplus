            if (x + 1 < dumpCols && x + 1 < lineSpan)

            }
        }

        ofs << "\n\n";
    }
}

// ------------------------------------------------------------
// 2Dテーブルをバイリニア補間でリサイズする関数
//
// small:
//   入力テーブル。
//   1次元vectorだが、中身は [y][x] の2次元データとして扱う。
//   index = y * smallWidth + x
//
// smallWidth / smallHeight:
//   入力側の幅・高さ
//
// bigWidth / bigHeight:
//   出力側の幅・高さ
//
// 戻り値:
//   bigWidth × bigHeight の uint16_t テーブル
//
// 注意:
//   12bit値なら 0〜4095 の範囲に収める。
// ------------------------------------------------------------
std::vector<std::uint16_t> resizeBilinear2D(
    const std::vector<std::uint16_t>& small,
    int smallWidth,
    int smallHeight,
    int bigWidth,
    int bigHeight
)
{
    std::vector<std::uint16_t> big(
        static_cast<std::size_t>(bigWidth * bigHeight)
    );

    for (int dstY = 0; dstY < bigHeight; ++dstY)
    {
        // ----------------------------------------------------
        // Big側のY座標が、Small側のどの位置に対応するかを求める。
        //
        // 例:
        //   dstY = 100
        //   srcY = 64.3
        //
        // これは「Small側の64行目と65行目の間」という意味。
        // ----------------------------------------------------
        double srcY =
            static_cast<double>(dstY) *
            static_cast<double>(smallHeight - 1) /
            static_cast<double>(bigHeight - 1);

        int y0 = static_cast<int>(std::floor(srcY));
        int y1 = std::min(y0 + 1, smallHeight - 1);

        // y方向の混ぜる割合
        // srcY=64.3 なら、64行目を70%、65行目を30%使うイメージ。
        double yRatio = srcY - static_cast<double>(y0);

        for (int dstX = 0; dstX < bigWidth; ++dstX)
        {
            // ------------------------------------------------
            // Big側のX座標が、Small側のどの位置に対応するかを求める。
            // ------------------------------------------------
            double srcX =
                static_cast<double>(dstX) *
                static_cast<double>(smallWidth - 1) /
                static_cast<double>(bigWidth - 1);

            int x0 = static_cast<int>(std::floor(srcX));
            int x1 = std::min(x0 + 1, smallWidth - 1);

            // x方向の混ぜる割合
            double xRatio = srcX - static_cast<double>(x0);

            // ------------------------------------------------
            // 周囲4点を取得する。
            //
            // q00: 左上
            // q10: 右上
            // q01: 左下
            // q11: 右下
            // ------------------------------------------------
            std::uint16_t q00 =
                small[static_cast<std::size_t>(y0 * smallWidth + x0)];

            std::uint16_t q10 =
                small[static_cast<std::size_t>(y0 * smallWidth + x1)];

            std::uint16_t q01 =
                small[static_cast<std::size_t>(y1 * smallWidth + x0)];

            std::uint16_t q11 =
                small[static_cast<std::size_t>(y1 * smallWidth + x1)];

            // ------------------------------------------------
            // まずX方向に補間する。
            //
            // top:
            //   上段の q00 と q10 を混ぜる
            //
            // bottom:
            //   下段の q01 と q11 を混ぜる
            // ------------------------------------------------
            double top =
                static_cast<double>(q00) * (1.0 - xRatio)
                +
                static_cast<double>(q10) * xRatio;

            double bottom =
                static_cast<double>(q01) * (1.0 - xRatio)
                +
                static_cast<double>(q11) * xRatio;

            // ------------------------------------------------
            // 次にY方向に補間する。
            //
            // top と bottom を混ぜて最終値にする。
            // ------------------------------------------------
            double value =
                top * (1.0 - yRatio)
                +
                bottom * yRatio;

            // ------------------------------------------------
            // 12bit値なら 0〜4095 に丸める。
            // uint8_tなら 0〜255 にする。
            // ------------------------------------------------
            int rounded = static_cast<int>(std::round(value));

            rounded = std::max(0, rounded);
            rounded = std::min(4095, rounded);

            big[static_cast<std::size_t>(dstY * bigWidth + dstX)] =
                static_cast<std::uint16_t>(rounded);
        }
    }

    return big;
}

// ============================================================
// 2Dテーブルの値アクセス用関数
//
// vectorを使っているが、内部的には
// table[y][x]
// と同じように扱う。
// ============================================================
std::uint16_t getValue(
    const std::vector<std::uint16_t>& table,
    int width,
    int x,
    int y
)
{
    return table[static_cast<std::size_t>(y * width + x)];
}
void setValue(
    std::vector<std::uint16_t>& table,
    int width,
    int x,
    int y,
    std::uint16_t value
)
{
    table[static_cast<std::size_t>(y * width + x)] = value;
}
// ============================================================
// CSV出力
//
// 2Dテーブルを、Excelなどで見やすいCSV形式に出力する。
// ============================================================
void writeCsv(
    const std::string& filePath,
    const std::vector<std::uint16_t>& table,
    int width,
    int height
)
{
    std::ofstream ofs(filePath);
    if (!ofs)
    {
        return;
    }
    for (int y = 0; y < height; ++y)
    {
        for (int x = 0; x < width; ++x)
        {
            ofs << getValue(table, width, x, y);
            if (x + 1 < width)
            {
                ofs << ",";
            }
        }
        ofs << "\n";
    }
}

// ============================================================
// 検証データ2: 4象限パターン
//
// 左上 = 0
// 右上 = 1
// 左下 = 2
// 右下 = 3
//
// 目的:
// - 上下左右が反転していないか
// - x/yを取り違えていないか
// - 最近傍補間の結果が見やすいか
// ============================================================
std::vector<std::uint16_t> createQuadrantPattern(
    int width,
    int height
)
{
    std::vector<std::uint16_t> table(
        static_cast<std::size_t>(width * height),
        0
    );
    for (int y = 0; y < height; ++y)
    {
        for (int x = 0; x < width; ++x)
        {
            bool right = x >= width / 2;
            bool bottom = y >= height / 2;
            std::uint16_t value = 0;
            if (!right && !bottom)
            {
                value = 0; // 左上
            }
            else if (right && !bottom)
            {
                value = 1; // 右上
            }
            else if (!right && bottom)
            {
                value = 2; // 左下
            }
            else
            {
                value = 3; // 右下
            }
            setValue(table, width, x, y, value);
        }
    }
    return table;
}

// ============================================================
// 最近傍補間
//
// Big側の各座標に対して、対応するSmall側の一番近い値を使う。
// 実装確認用として最初に使う。
// ============================================================
std::vector<std::uint16_t> resizeNearest2D(
    const std::vector<std::uint16_t>& small,
    int smallWidth,
    int smallHeight,
    int bigWidth,
    int bigHeight
)
{
    std::vector<std::uint16_t> big(
        static_cast<std::size_t>(bigWidth * bigHeight),
        0
    );
    for (int dstY = 0; dstY < bigHeight; ++dstY)
    {
        int srcY =
            dstY
            * smallHeight
            / bigHeight;
        if (srcY >= smallHeight)
        {
            srcY = smallHeight - 1;
        }
        for (int dstX = 0; dstX < bigWidth; ++dstX)
        {
            int srcX =
                dstX
                * smallWidth
                / bigWidth;
            if (srcX >= smallWidth)
            {
                srcX = smallWidth - 1;
            }
            std::uint16_t value =
                getValue(
                    small,
                    smallWidth,
                    srcX,
                    srcY
                );
            setValue(
                big,
                bigWidth,
                dstX,
                dstY,
                value
            );
        }
    }
    return big;
}
// ============================================================
// バイリニア補間
//
// 周囲4点を混ぜて値を作る。
// シェーディング補正のような滑らかな補正マップ向け。
// ============================================================
std::vector<std::uint16_t> resizeBilinear2D(
    const std::vector<std::uint16_t>& small,
    int smallWidth,
    int smallHeight,
    int bigWidth,
    int bigHeight
)
{
    std::vector<std::uint16_t> big(
        static_cast<std::size_t>(bigWidth * bigHeight),
        0
    );
    for (int dstY = 0; dstY < bigHeight; ++dstY)
    {
        double srcY = 0.0;
        if (bigHeight == 1)
        {
            srcY = 0.0;
        }
        else
        {
            // 端と端が対応するようにする
            // dstY=0             -> srcY=0
            // dstY=bigHeight-1   -> srcY=smallHeight-1
            srcY =
                static_cast<double>(dstY)
                * static_cast<double>(smallHeight - 1)
                / static_cast<double>(bigHeight - 1);
        }
        int y0 = static_cast<int>(std::floor(srcY));
        int y1 = std::min(y0 + 1, smallHeight - 1);
        double yRatio =
            srcY - static_cast<double>(y0);
        for (int dstX = 0; dstX < bigWidth; ++dstX)
        {
            double srcX = 0.0;
            if (bigWidth == 1)
            {
                srcX = 0.0;
            }
            else
            {
                srcX =
                    static_cast<double>(dstX)
                    * static_cast<double>(smallWidth - 1)
                    / static_cast<double>(bigWidth - 1);
            }
            int x0 = static_cast<int>(std::floor(srcX));
            int x1 = std::min(x0 + 1, smallWidth - 1);
            double xRatio =
                srcX - static_cast<double>(x0);
            std::uint16_t q00 =
                getValue(small, smallWidth, x0, y0); // 左上
            std::uint16_t q10 =
                getValue(small, smallWidth, x1, y0); // 右上
            std::uint16_t q01 =
                getValue(small, smallWidth, x0, y1); // 左下
            std::uint16_t q11 =
                getValue(small, smallWidth, x1, y1); // 右下
            // x方向に補間
            double top =
                static_cast<double>(q00) * (1.0 - xRatio)
                +
                static_cast<double>(q10) * xRatio;
            double bottom =
                static_cast<double>(q01) * (1.0 - xRatio)
                +
                static_cast<double>(q11) * xRatio;
            // y方向に補間
            double value =
                top * (1.0 - yRatio)
                +
                bottom * yRatio;
            int rounded =
                static_cast<int>(std::round(value));
            // 12bit想定なら0〜4095に収める
            rounded = std::max(0, rounded);
            rounded = std::min(4095, rounded);
            setValue(
                big,
                bigWidth,
                dstX,
                dstY,
                static_cast<std::uint16_t>(rounded)
            );
        }
    }
    return big;
}

欠陥画素テーブルが「x, y 座標リスト」なら、主な手法はスケール変換です。

ただし、厳密には候補がいくつかあります。

欠陥画素テーブルの変換手法

1. 単純スケール変換

一番基本。

newX = oldX * bigWidth  / smallWidth;
newY = oldY * bigHeight / smallHeight;

意味は、

Small画像上の相対位置をBig画像上へ移す

です。

⸻

2. 端点対応スケール変換

欠陥画素のような「座標」では、こちらの方が自然なことが多いです。

newX = oldX * (bigWidth  - 1) / (smallWidth  - 1);
newY = oldY * (bigHeight - 1) / (smallHeight - 1);

これだと、

Smallの左端 → Bigの左端
Smallの右端 → Bigの右端
Smallの上端 → Bigの上端
Smallの下端 → Bigの下端

が保証されます。

欠陥画素では、まずこの方式を第一候補にしてよいです。

⸻

3. 丸め方式の選択

スケール後に小数が出る場合があります。

例:

oldX = 100
newX = 198.75

このとき、どれにするか。

切り捨て floor → 198
四捨五入 round → 199
切り上げ ceil → 199

欠陥画素では、round の方が自然なことが多いです。
理由は「一番近いBig側の画素」に移したいからです。

C++ならこう。

newX = round(oldX * scaleX);
newY = round(oldY * scaleY);

ただし、既存仕様が整数除算なら、それに合わせる必要があります。

⸻

4. offset込み変換

カメラ画像の切り出し位置や有効領域が違う場合は、倍率だけでは足りません。

newX = oldX * scaleX + offsetX;
newY = oldY * scaleY + offsetY;

例えば、

Smallは画像全体
Bigは左に100pxずれた有効領域

のような場合、offsetが必要です。

⸻

5. affine変換

もし、単なる拡大縮小だけではなく、

回転
平行移動
せん断

があるなら、Affine変換になります。

newX = a * oldX + b * oldY + tx;
newY = c * oldX + d * oldY + ty;

ただし、欠陥画素テーブル単体では、まずここまでは不要な可能性が高いです。
現場から「欠陥画素にも回転・歪み補正を反映する」と言われていないなら、最初からAffineに行くべきではありません。

⸻

6. 除外・クリップ

変換後の座標が範囲外になる場合があります。

その場合はどちらか。

範囲外なら除外する
範囲内に丸め込む

例:

if (newX >= bigWidth) {
    newX = bigWidth - 1;
}

ただし、欠陥画素としては「存在しない座標」を無理やり端に寄せるより、範囲外なら除外の方が自然な場合もあります。

⸻

現実的な第一案

今回の初期方針としてはこれがよいです。

方式:
端点対応スケール変換
丸め:
round
範囲外:
除外またはclip。まずはclipより除外方針を確認
num:
変換後に有効な欠陥画素数を再設定

コードイメージ。

double scaleX =
    static_cast<double>(bigWidth - 1) /
    static_cast<double>(smallWidth - 1);
double scaleY =
    static_cast<double>(bigHeight - 1) /
    static_cast<double>(smallHeight - 1);
newX = static_cast<int>(std::round(oldX * scaleX));
newY = static_cast<int>(std::round(oldY * scaleY));

⸻

まとめると、

通常の欠陥画素テーブル
→ スケール変換が中心
有効領域ズレあり
→ scale + offset
回転や歪み補正あり
→ affine / calibration変換
変換後範囲外
→ 除外 or clip

なので、まずは 端点対応スケール変換 + round + 範囲チェック を実装・検証するのが妥当です。
